definition{
  name: vnxIndent,
  parameters: [line : Bytes, i : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (if ((Foundation.Tar::byteOr line i) =? 32) then (vnxIndent line (i + 1) rest) else i)),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: vnxKind,
  parameters: [line : Bytes],
  result: Nat,
  body: (let n : Nat := (vnxIndent line 0 (GenIcf::icfLen line)); (if (GenIcf::icfStarts (GenIcf::icfSlice line n (GenIcf::icfLen line)) (GenIcf::icfJoin 0x22 (GenIcf::icfJoin (GenIcf::icfText "id") (GenIcf::icfJoin 0x22 (GenIcf::icfJoin (GenIcf::icfText ": ") 0x22))))) then 1 else (if ((n =? 2) && (((Foundation.Tar::byteOr line 2) =? 34) && (GenIcf::icfEnds line (GenIcf::icfJoin 0x22 (GenIcf::icfText ": ["))))) then 2 else 0))),
  axioms: [Quot.sound, propext]
}

definition{
  name: vnxText,
  parameters: [line : Bytes],
  result: Bytes,
  body: (let n : Nat := (vnxIndent line 0 (GenIcf::icfLen line)); (if ((vnxKind line) =? 1) then (GenIcf::icfSlice line (n + 7) (GenIcf::icfStopAt line (n + 7) (GenIcf::icfLen line) 34 (GenIcf::icfLen line))) else (GenIcf::icfSlice line 3 (prim subtract (GenIcf::icfLen line) 4 : Nat)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: vnxName,
  parameters: [line : Bytes],
  result: Bytes,
  body: (GenIcf::icfJoin (GenIcf::icfText (if ((vnxKind line) =? 2) then "vocabTokensOf" else "vocabTokens")) (vnxText line)),
  axioms: [Quot.sound, propext]
}

definition{
  name: vnxRow,
  parameters: [line : Bytes],
  result: Bytes,
  body: (GenIcf::icfJoin (GenIcf::icfJoin (GenIcf::icfText "(cons (record VocabToken {mark := ") (GenIcf::icfText (if ((vnxKind line) =? 2) then "true" else "false"))) (GenIcf::icfJoin (GenIcf::icfText ", text := ") (GenIcf::icfJoin (GenIcf::icfString (vnxText line)) (GenIcf::icfText "}) ")))),
  axioms: [Quot.sound, propext]
}

definition{
  name: vnxCloses,

  recursive_argument: n
}

definition{
  name: vnxClose,

  result: Bytes,
  body: (GenIcf::icfJoin (GenIcf::icfText "(nil VocabToken)") (vnxCloses k))
}

definition{
  name: vnxOpen,

  result: Bytes,
  body: (GenIcf::icfJoin (GenIcf::icfText "definition{\n  name: ") (GenIcf::icfJoin (vnxName line) (GenIcf::icfText ",\n  parameters: [],\n  result: (List VocabToken),\n  body: "))),
  axioms: [Quot.sound, propext]
}

definition{
  name: vnxHead,
  parameters: [unused : Nat],
  result: Bytes,
  body: (GenIcf::icfJoin (GenIcf::icfJoin (GenIcf::icfJoin 0x5c (GenIcf::icfJoin (GenIcf::icfText "begin{lexlean}{VocabNames}\n") 0x5c)) (GenIcf::icfJoin (GenIcf::icfJoin (GenIcf::icfText "useglossary{lexlean.std.bool@1.2.0}\n") 0x5c) (GenIcf::icfJoin (GenIcf::icfText "useglossary{lexlean.std.nat@1.2.0}\n") 0x5c))) (GenIcf::icfJoin (GenIcf::icfJoin (GenIcf::icfText "useglossary{prism.arch@1.0.0}\n") (GenIcf::icfJoin 0x5c (GenIcf::icfText "title{Schema.org register}\n"))) (GenIcf::icfJoin (GenIcf::icfJoin 0x5c (GenIcf::icfText "begin{semanticmodule}\nstructure{\n  name: VocabToken,\n  type_parameters: [],\n  parameters: [],\n  fields: [{name: mark, type: Bool}, {name: text, type: String}]\n}\n\ndefinition{\n  name")) (GenIcf::icfJoin (GenIcf::icfText ": vocabAppendTokens,\n  parameters: [front : (List VocabToken), back : (List VocabToken)],\n  result: (List VocabToken),\n  body: (match front\n    | List.nil => back\n    | List.cons i") (GenIcf::icfText "tem rest => (cons item (vocabAppendTokens rest back))),\n  recursive_argument: front\n}\n\n")))))
}

definition{
  name: vnxFrom,
  parameters: [s : Bytes, i : Nat, k : Nat, acc : Bytes, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (if ((GenIcf::icfLen s) <=? i) then (if (k =? 0) then acc else (GenIcf::icfJoin acc (vnxClose k))) else (let e : Nat := (GenIcf::icfStopAt s i (GenIcf::icfLen s) 10 (GenIcf::icfLen s)); (let line : Bytes := (GenIcf::icfLineOf s i e); (if ((i =? 0) && (Foundation.Tar::sameBytes line (GenIcf::icfText "{"))) then (vnxFrom s (e + 1) k (GenIcf::icfJoin acc (vnxHead 0)) rest) else (if ((vnxKind line) =? 0) then (vnxFrom s (e + 1) k acc rest) else (let opened : Bytes := (if (k =? 0) then (GenIcf::icfJoin acc (vnxOpen line)) else acc); (if ((k + 1) =? 50) then (vnxFrom s (e + 1) 0 (GenIcf::icfJoin opened (GenIcf::icfJoin (vnxRow line) (vnxClose 50))) rest) else (vnxFrom s (e + 1) (k + 1) (GenIcf::icfJoin opened (vnxRow line)) rest))))))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: vnxNamesFrom,
  parameters: [s : Bytes, i : Nat, k : Nat, acc : Bytes, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (if ((GenIcf::icfLen s) <=? i) then (if (k =? 0) then acc else (GenIcf::icfJoin acc (GenIcf::icfJoin (GenIcf::icfDigits k) (GenIcf::icfText "\n")))) else (let e : Nat := (GenIcf::icfStopAt s i (GenIcf::icfLen s) 10 (GenIcf::icfLen s)); (let line : Bytes := (GenIcf::icfLineOf s i e); (if ((i =? 0) && (Foundation.Tar::sameBytes line (GenIcf::icfText "{"))) then (vnxNamesFrom s (e + 1) k (GenIcf::icfJoin acc 0x) rest) else (if ((vnxKind line) =? 0) then (vnxNamesFrom s (e + 1) k acc rest) else (let opened : Bytes := (if (k =? 0) then (GenIcf::icfJoin acc (GenIcf::icfJoin (vnxName line) (GenIcf::icfText " "))) else acc); (if ((k + 1) =? 50) then (vnxNamesFrom s (e + 1) 0 (GenIcf::icfJoin opened (GenIcf::icfText "50\n")) rest) else (vnxNamesFrom s (e + 1) (k + 1) (GenIcf::icfJoin opened 0x) rest))))))))),

  recursive_argument: fuel
}

  result: Bytes,
  body: (match fuel
    | Nat.zero => (GenIcf::icfText "(nil VocabToken)")
    | Nat.succ rest => (if (((GenIcf::icfLen s) <=? i) || (left =? 0)) then (GenIcf::icfText "(nil VocabToken)") else (let e : Nat := (GenIcf::icfStopAt s i (GenIcf::icfLen s) 10 (GenIcf::icfLen s)); (let sp : Nat := (GenIcf::icfStopAt s i e 32 e); (GenIcf::icfJoin (GenIcf::icfJoin (GenIcf::icfText "(vocabAppendTokens (") (GenIcf::icfSlice s i sp)) (GenIcf::icfJoin (GenIcf::icfText ") ") (GenIcf::icfJoin (vnxChain s (e + 1) (prim subtract left 1 : Nat) rest) (GenIcf::icfText ")")))))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: vnxSkip,
  parameters: [s : Bytes, i : Nat, left : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (if (((GenIcf::icfLen s) <=? i) || (left =? 0)) then i else (vnxSkip s ((GenIcf::icfStopAt s i (GenIcf::icfLen s) 10 (GenIcf::icfLen s)) + 1) (prim subtract left 1 : Nat) rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: vnxParts,
  parameters: [s : Bytes, i : Nat, p : Nat, acc : Bytes, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (if ((GenIcf::icfLen s) <=? i) then acc else (vnxParts s (vnxSkip s i 30 30) (p + 1) (GenIcf::icfJoin (GenIcf::icfJoin acc (GenIcf::icfJoin (GenIcf::icfText "definition{\n  name: ") (GenIcf::icfJoin (GenIcf::icfJoin (GenIcf::icfText "vocabTokensPart") (GenIcfLean::ilxDecimal p 20)) (GenIcf::icfText ",\n  parameters: [],\n  result: (List VocabToken),\n  body: ")))) (GenIcf::icfJoin (vnxChain s i 30 30) (GenIcf::icfText "\n}\n\n"))) rest))),

definition{
  name: vnxPartChain,

  body: (match fuel
    | Nat.zero => (GenIcf::icfText "(nil VocabToken)")
    | Nat.succ rest => (if (count <=? p) then (GenIcf::icfText "(nil VocabToken)") else (GenIcf::icfJoin (GenIcf::icfJoin (GenIcf::icfText "(vocabAppendTokens (vocabTokensPart") (GenIcfLean::ilxDecimal p 20)) (GenIcf::icfJoin (GenIcf::icfText ") ") (GenIcf::icfJoin (vnxPartChain (p + 1) count rest) (GenIcf::icfText ")")))))),
  recursive_argument: fuel
