\begin{lexlean}{GenWcag}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Foundation.Json}
\importmodule{Foundation.Tar}
\importmodule{Foundation.Utf8}
\title{Accessibility criteria generator}
\begin{semanticmodule}
definition{
  name: u,
  parameters: [value : String],
  result: Bytes,
  body: (Foundation.Utf8::encode value)
}

definition{
  name: cat,
  parameters: [parts : (List Bytes)],
  result: Bytes,
  body: (match parts
    | List.nil => 0x
    | List.cons head rest => (Foundation.Bytes::appendBytes head (cat rest))),
  recursive_argument: parts
}

definition{
  name: quoted,
  parameters: [value : Bytes],
  result: Bytes,
  body: (cat (cons 0x22 (cons value (cons 0x22 (nil Bytes)))))
}

definition{
  name: startsAt,
  parameters: [s : Bytes, i : Nat, pattern : Bytes],
  result: Bool,
  body: (Foundation.Tar::sameBytes (Foundation.Json::span s i (i + (Foundation.Bytes::byteLength pattern))) pattern),
  axioms: [propext]
}

definition{
  name: wcagLineEndAt,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (if ((Foundation.Bytes::byteLength s) <? (i + 1)) then i else (if ((Foundation.Tar::byteOr s i) =? 10) then i else (wcagLineEndAt s (i + 1) rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: wcagBackslashFrom,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Bool,
  body: (match fuel
    | Nat.zero => false
    | Nat.succ rest => (if ((Foundation.Bytes::byteLength s) <? (i + 1)) then false else (if ((Foundation.Tar::byteOr s i) =? 92) then true else (wcagBackslashFrom s (i + 1) rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: wcagMemberOf,
  parameters: [line : Bytes, key : String],
  result: (Option Bytes),
  body: (let head : Bytes := (cat (cons (u "              ") (cons 0x22 (cons (u key) (cons 0x223a2022 (nil Bytes)))))); (let n : Nat := (Foundation.Bytes::byteLength line); (let h : Nat := (Foundation.Bytes::byteLength head); (if (startsAt line 0 head) then (let stop : Nat := (if ((Foundation.Tar::byteOr line (prim subtract n 1 : Nat)) =? 44) then (prim subtract n 2 : Nat) else (prim subtract n 1 : Nat)); (if ((h <? (stop + 1)) && ((Foundation.Tar::byteOr line stop) =? 34)) then (ctor Option.some [Bytes] (Foundation.Json::span line h stop)) else (ctor Option.none [Bytes]))) else (ctor Option.none [Bytes]))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: wcagFact,
  parameters: [key : String, value : Bytes],
  result: Bytes,
  body: (if (wcagBackslashFrom value 0 (Foundation.Bytes::byteLength value)) then (cat (cons (u "escaped") (cons 0x09 (cons value (cons 0x0a (nil Bytes)))))) else (cat (cons (u key) (cons 0x09 (cons value (cons 0x0a (nil Bytes))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: wcagFactOf,
  parameters: [line : Bytes],
  result: Bytes,
  body: (match (wcagMemberOf line "id")
    | Option.some v => (wcagFact "id" v)
    | Option.none => (match (wcagMemberOf line "num")
      | Option.some v => (wcagFact "num" v)
      | Option.none => (match (wcagMemberOf line "handle")
        | Option.some v => (wcagFact "handle" v)
        | Option.none => (match (wcagMemberOf line "level")
          | Option.some v => (wcagFact "level" v)
          | Option.none => 0x)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: wcagFactsFrom,
  parameters: [s : Bytes, i : Nat, acc : Bytes, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (let n : Nat := (Foundation.Bytes::byteLength s); (if (n <? (i + 1)) then acc else (let e : Nat := (wcagLineEndAt s i n); (wcagFactsFrom s (e + 1) (Foundation.Bytes::appendBytes acc (wcagFactOf (Foundation.Json::span s i e))) rest))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: wcagFacts,
  parameters: [archive : Bytes],
  result: (Option Bytes),
  body: (match (Foundation.Tar::find archive (u "app/specs/standards/upstream/wcag.json"))
    | Option.none => (ctor Option.none [Bytes])
    | Option.some piece => (ctor Option.some [Bytes] (wcagFactsFrom piece 0 0x (Foundation.Bytes::byteLength piece)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: wcagValueOf,
  parameters: [line : Bytes, key : String],
  result: (Option Bytes),
  body: (let head : Bytes := (Foundation.Bytes::appendBytes (u key) 0x09); (if (startsAt line 0 head) then (ctor Option.some [Bytes] (Foundation.Json::span line (Foundation.Bytes::byteLength head) (Foundation.Bytes::byteLength line))) else (ctor Option.none [Bytes]))),
  axioms: [propext]
}

definition{
  name: wcagClaimed,
  parameters: [level : Bytes],
  result: Bool,
  body: ((Foundation.Tar::sameBytes level (u "A")) || (Foundation.Tar::sameBytes level (u "AA"))),
  axioms: [propext]
}

definition{
  name: wcagMember,
  parameters: [name : String, value : Bytes, final : Bool],
  result: Bytes,
  body: (cat (cons (u "  ") (cons (quoted (u name)) (cons (u ": ") (cons (quoted value) (cons (if final then 0x0a else 0x2c0a) (nil Bytes)))))))
}

definition{
  name: wcagCriterion,
  parameters: [anchor : Bytes, num : Bytes, handle : Bytes, level : Bytes],
  result: Bytes,
  body: (cat (cons (u " {") (cons 0x0a (cons (wcagMember "id" (Foundation.Bytes::appendBytes (u "wcag-2.2-") num) false) (cons (wcagMember "kind" (Foundation.Bytes::appendBytes (u "success criterion, level ") level) false) (cons (wcagMember "ref" (Foundation.Bytes::appendBytes (u "https://www.w3.org/TR/WCAG22/#") anchor) false) (cons (wcagMember "text" handle true) (cons (u " }") (nil Bytes)))))))))
}

definition{
  name: wcagCriteriaFrom,
  parameters: [s : Bytes, i : Nat, lead : Bool, acc : Bytes, fuel : Nat],
  result: (Option Bytes),
  body: (match fuel
    | Nat.zero => (ctor Option.none [Bytes])
    | Nat.succ rest => (let n : Nat := (Foundation.Bytes::byteLength s); (if (n <? (i + 1)) then (ctor Option.some [Bytes] acc) else (let e1 : Nat := (wcagLineEndAt s i n); (let e2 : Nat := (wcagLineEndAt s (e1 + 1) n); (let e3 : Nat := (wcagLineEndAt s (e2 + 1) n); (let e4 : Nat := (wcagLineEndAt s (e3 + 1) n); (match (wcagValueOf (Foundation.Json::span s i e1) "id")
      | Option.none => (ctor Option.none [Bytes])
      | Option.some anchor => (match (wcagValueOf (Foundation.Json::span s (e1 + 1) e2) "num")
        | Option.none => (ctor Option.none [Bytes])
        | Option.some num => (match (wcagValueOf (Foundation.Json::span s (e2 + 1) e3) "handle")
          | Option.none => (ctor Option.none [Bytes])
          | Option.some handle => (match (wcagValueOf (Foundation.Json::span s (e3 + 1) e4) "level")
            | Option.none => (ctor Option.none [Bytes])
            | Option.some level => (if (wcagClaimed level) then (wcagCriteriaFrom s (e4 + 1) false (Foundation.Bytes::appendBytes acc (Foundation.Bytes::appendBytes (if lead then 0x else 0x2c0a) (wcagCriterion anchor num handle level))) rest) else (wcagCriteriaFrom s (e4 + 1) lead acc rest))))))))))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: wcagGenerate,
  parameters: [archive : Bytes],
  result: (Option Bytes),
  body: (match (Foundation.Tar::find archive (u ".gen/wcag-facts"))
    | Option.none => (ctor Option.none [Bytes])
    | Option.some s => (match (wcagCriteriaFrom s 0 true 0x (Foundation.Bytes::byteLength s))
      | Option.none => (ctor Option.none [Bytes])
      | Option.some body => (ctor Option.some [Bytes] (cat (cons (u "[") (cons 0x0a (cons body (cons 0x0a (cons (u "]") (cons 0x0a (nil Bytes))))))))))),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
