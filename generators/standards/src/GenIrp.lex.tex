definition{
  name: irpFind,
  parameters: [s : Bytes, i : Nat, pattern : Bytes, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => (Foundation.Bytes::byteLength s)
    | Nat.succ rest => (if ((Foundation.Bytes::byteLength s) <? (i + 1)) then (Foundation.Bytes::byteLength s) else (if (GenWcag::startsAt s i pattern) then i else (irpFind s (i + 1) pattern rest)))),
  axioms: [propext],
  recursive_argument: fuel
}

definition{
  name: irpAt,
  parameters: [s : Bytes, i : Nat, pattern : String],
  result: Nat,
  body: (irpFind s i (GenWcag::u pattern) (Foundation.Bytes::byteLength s)),
  axioms: [propext]
}

definition{
  name: irpHas,
  parameters: [s : Bytes, i : Nat, pattern : String],
  result: Bool,
  body: ((irpAt s i pattern) <? (Foundation.Bytes::byteLength s)),
  axioms: [propext]
}

definition{
  name: irpValue,
  parameters: [s : Bytes, i : Nat, key : Bytes],
  result: Bytes,
  body: (let a : Nat := ((irpFind s i key (Foundation.Bytes::byteLength s)) + (Foundation.Bytes::byteLength key)); (Foundation.Json::span s a (irpFind s a 0x22 (Foundation.Bytes::byteLength s)))),
  axioms: [propext]
}

definition{
  name: irpIsSpace,
  parameters: [b : Nat],
  result: Bool,
  body: ((b =? 32) || ((b =? 10) || ((b =? 13) || (b =? 9))))
}

  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: irpKind,

definition{
  name: irpHead,
  parameters: [s : Bytes, a : Nat, c : Nat, e : Nat],
  result: Bytes,
  body: (GenWcag::cat (cons 0x20207b226964223a20227733632d7363786d6c2d312e302d (cons (irpValue s a 0x69643d22) (cons 0x222c2022726566223a2022 (cons (irpValue s a 0x737065636e756d3d22) (cons 0x222c202274657874223a2022 (cons (irpTextFrom s (c + 9) e false 0x (Foundation.Bytes::byteLength s)) (cons 0x22 (nil Bytes))))))))),
  axioms: [Quot.sound, propext]

definition{
  name: irpEntriesFrom,
  parameters: [s : Bytes, i : Nat, lead : Bool, acc : Bytes, fuel : Nat],
  result: (Option Bytes),
  body: (match fuel
    | Nat.zero => (ctor Option.none [Bytes])
    | Nat.succ rest => (let n : Nat := (Foundation.Bytes::byteLength s); (let a : Nat := (irpAt s i "<assert "); (if (n <?   axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: irpPiece,
  parameters: [s : Bytes],
  result: (Option Bytes),
  body: (let opening : Bool := (GenWcag::startsAt s 0 (GenWcag::u "<?xml")); (let close : Nat := (irpAt s 0 "</assert>"); (let tail : Bool := (close <?     | Option.some body => (ctor Option.some [Bytes] (if (irpHas s 0 "</assertions>") then (Foundation.Bytes::appendBytes body 0x0a5d0a) else body)))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: irpGenerate,
  parameters: [archive : Bytes],
  result: (Option Bytes),
  body: (match (Foundation.Tar::find archive (GenWcag::u "app/specs/standards/upstream/scxml-irp-manifest.xml"))
    | Option.none => (ctor Option.none [Bytes])
    | Option.some piece => (irpPiece piece)),
  axioms: [Quot.sound, propext]
}
