definition{
  name: tkHex,
  parameters: [c : Nat],
  result: Nat,
  body: (if ((47 <? c) && (c <? 58)) then (prim subtract c 48 : Nat) else (if ((96 <? c) && (c <? 103)) then (prim subtract c 87 : Nat) else (if ((64 <? c) && (c <? 71)) then (prim subtract c 55 : Nat) else 16)))
}

definition{
  name: tkHexOkFrom,
  parameters: [h : Bytes, i : Nat, fuel : Nat],
  result: Bool,
  body: (match fuel
    | Nat.zero => true
    | Nat.succ rest => (if ((tkHex (Foundation.Tar::byteOr h i)) <? 16) then (tkHexOkFrom h (i + 1) rest) else false)),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: tkHexOk,
  parameters: [h : Bytes],
  result: Bool,
  body: ((((Foundation.Bytes::byteLength h) =? 7) && ((Foundation.Tar::byteOr h 0) =? 35)) && (tkHexOkFrom h 1 6)),
  axioms: [Quot.sound, propext]
}

definition{
  name: tkPad,
  parameters: [v : Nat, width : Nat],
  result: Bytes,
  body: (match width
    | Nat.zero => 0x
    | Nat.succ rest => (let d : Nat := (prim remainder v 10 0 : Nat); (Foundation.Bytes::appendBytes (tkPad (prim quotient v 10 0 : Nat) rest) (Foundation.Json::span 0x30313233343536373839 d (d + 1))))),
  recursive_argument: width
}

definition{
  name: tkTrim,
  parameters: [d : Bytes, e : Nat, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => (Foundation.Json::span d 0 e)
    | Nat.succ rest => (if ((1 <? e) && ((Foundation.Tar::byteOr d (prim subtract e 1 : Nat)) =? 48)) then (tkTrim d (prim subtract e 1 : Nat) rest) else (Foundation.Json::span d 0 e))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: tkComponent,
  parameters: [h : Bytes, i : Nat],
  result: Bytes,

  axioms: [Quot.sound, propext]
}

definition{
  name: tkJoin,
  parameters: [acc : Bytes, item : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength acc) =? 0) then item else (GenLook::lookCat (cons acc (cons 0x2c0a (cons item (nil Bytes))))))
}

definition{
  name: tkTonesFrom,
  parameters: [s : Bytes, tones : (Option Nat), theme : Bytes, n : Nat, acc : Bytes, fuel : Nat],
  result: (Option Bytes),
  body: (match fuel
    | Nat.zero => (ctor Option.none [Bytes])
    | Nat.succ rest => (match (Foundation.Json::elementAt s tones n)
      | Option.none => (ctor Option.some [Bytes] acc)
      | Option.some i => (let t : (Option Nat) := (ctor Option.some [Nat] i); (let h : Bytes := (Foundation.Json::textAt s (Foundation.Json::memberAt s t theme)); (if (tkHexOk h) then (tkTonesFrom s tones theme (n + 1) (tkJoin acc (GenLook::lookCat (cons 0x202022 (cons (GenLook::lookTextOf s t "key") (cons 0x223a207b222474797065223a2022636f6c6f72222c20222476616c7565223a20 (cons (tkColor h) (cons 0x7d (nil Bytes)))))))) rest) else (ctor Option.none [Bytes])))))),
  axioms: [Quot.sound, propext],

definition{
  name: tkThemesFrom,
  parameters: [s : Bytes, look : (Option Nat), n : Nat, acc : Bytes, fuel : Nat],
  result: (Option Bytes),
  body: (match fuel
    | Nat.zero => (ctor Option.none [Bytes])
    | Nat.succ rest => (match (Foundation.Json::elementAt s (Foundation.Json::memberAt s look (GenLook::lookU "themes")) n)
      | Option.none => (ctor Option.some [Bytes] acc)
      | Option.some i => (let theme : Bytes := (Foundation.Json::textAt s (ctor Option.some [Nat] i)); (match (tkTonesFrom s (Foundation.Json::memberAt s look (GenLook::lookU "tones")) theme 0 0x (Foundation.Bytes::byteLength s))
        | Option.none => (ctor Option.none [Bytes])
        | Option.some tones => (tkThemesFrom s look (n + 1) (GenLook::lookCat (cons acc (cons 0x2022 (cons theme (cons 0x223a207b0a (cons tones (cons 0x0a207d2c0a (nil Bytes)))))))) rest))))),

  recursive_argument: fuel
}

    | Nat.zero => i

  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}
