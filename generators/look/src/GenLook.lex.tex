definition{
  name: lookU,
  parameters: [value : String],
  result: Bytes,
  body: (Foundation.Utf8::encode value)

definition{
  name: lookCat,
  parameters: [parts : (List Bytes)],
  result: Bytes,
  body: (match parts
    | List.nil => 0x
    | List.cons head rest => (Foundation.Bytes::appendBytes head (lookCat rest))),
  recursive_argument: parts
}

definition{
  name: lookCompactFrom,
  parameters: [s : Bytes, i : Nat, stop : Nat, inString : Bool, acc : Bytes, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (if (stop <? (i + 1)) then acc else (let c : Nat := (Foundation.Tar::byteOr s i); (if inString then (if (c =? 92) then (lookCompactFrom s (i + 2) stop true (Foundation.Bytes::appendBytes acc (Foundation.Json::span s i (i + 2))) rest) else (lookCompactFrom s (i + 1) stop (if (c =? 34) then false else true) (Foundation.Bytes::appendBytes acc (Foundation.Json::span s i (i + 1))) rest)) else (if (c =? 34) then (lookCompactFrom s (i + 1) stop true (Foundation.Bytes::appendBytes acc 0x22) rest) else (if ((c =? 32) || ((c =? 10) || ((c =? 9) || (c =? 13)))) then (lookCompactFrom s (i + 1) stop false acc rest) else (if (c =? 44) then (lookCompactFrom s (i + 1) stop false (Foundation.Bytes::appendBytes acc 0x2c20) rest) else (if (c =? 58) then (lookCompactFrom s (i + 1) stop false (Foundation.Bytes::appendBytes acc 0x3a20) rest) else (lookCompactFrom s (i + 1) stop false (Foundation.Bytes::appendBytes acc (Foundation.Json::span s i (i + 1))) rest))))))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: lookCompact,
  parameters: [s : Bytes, pos : (Option Nat)],
  result: Bytes,
  body: (match pos
    | Option.none => 0x
    | Option.some i => (lookCompactFrom s i (Foundation.Json::valueEnd s i) false 0x (Foundation.Bytes::byteLength s))),
  axioms: [Quot.sound, propext]
}

definition{
  name: lookTextOf,
  parameters: [s : Bytes, pos : (Option Nat), key : String],
  result: Bytes,
  body: (Foundation.Json::textAt s (Foundation.Json::memberAt s pos (lookU key))),
  axioms: [Quot.sound, propext]
}

definition{
  name: lookHasEvery,
  parameters: [s : Bytes, keys : (List String)],
  result: Bool,
  body: (match keys
    | List.nil => true
    | List.cons head rest => (match (Foundation.Json::memberAt s (Foundation.Json::root s) (lookU head))
      | Option.none => false
      | Option.some v => (lookHasEvery s rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: keys
}

definition{
  name: lookJoinSemi,
  parameters: [acc : Bytes, item : Bytes],
  result: Bytes,
  body: (if ((Foundation.Bytes::byteLength acc) =? 0) then item else (lookCat (cons acc (cons 0x3b20 (cons item (nil Bytes))))))
}

definition{

definition{
  name: lookFindFrom,
  parameters: [s : Bytes, pattern : Bytes, i : Nat, fuel : Nat],
  result: (Option Nat),
  body: (match fuel
    | Nat.zero => (ctor Option.none [Nat])
    | Nat.succ rest => (if ((Foundation.Bytes::byteLength s) <?   axioms: [propext],
  recursive_argument: fuel
}
