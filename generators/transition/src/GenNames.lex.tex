definition{
  name: gnFindFrom,
  parameters: [s : Bytes, pat : Bytes, i : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => (GenPatch::jpLen s)
    | Nat.succ rest => (if ((GenPatch::jpLen s) <? (i + (GenPatch::jpLen pat))) then (GenPatch::jpLen s) else (if (GenPatch::jpSame (GenPatch::jpSpan s i (i + (GenPatch::jpLen pat))) pat) then i else (gnFindFrom s pat (i + 1) rest)))),
  axioms: [propext],
  recursive_argument: fuel
}

definition{
  name: gnFind,
  parameters: [s : Bytes, pat : Bytes, i : Nat],
  result: Nat,
  body: (gnFindFrom s pat i ((GenPatch::jpLen s) + 1)),
  axioms: [propext]
}

definition{
  name: gnEndFrom,
  parameters: [s : Bytes, i : Nat, wide : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (if ((GenPatch::jpLen s) <? (i + 1)) then i else (let c : Nat := (GenPatch::jpAt s i); (if ((c =? 10) || ((c =? 32) && ((wide =? 0) || ((GenPatch::jpAt s (i + 1)) =? 32)))) then i else (gnEndFrom s (i + 1) wide rest))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: gnTok,
  parameters: [s : Bytes, i : Nat, wide : Nat],
  result: Bytes,
  body: (GenPatch::jpSpan s i (gnEndFrom s i wide ((GenPatch::jpLen s) + 1))),
  axioms: [Quot.sound, propext]
}

definition{
  name: gnSkip,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (if (((GenPatch::jpAt s i) =? 32) || ((GenPatch::jpAt s i) =? 124)) then (gnSkip s (i + 1) rest) else i)),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: gnNextLine,
  parameters: [s : Bytes, i : Nat],
  result: Nat,
  body: ((gnFind s 0x0a i) + 1),
  axioms: [propext]
}

definition{
  name: gnHexRun,
  parameters: [s : Bytes, i : Nat, acc : Bytes, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (let c : Nat := (GenPatch::jpAt s i); (if (c =? 32) then (gnHexRun s (i + 1) acc rest) else (if ((c =? 10) || (c =? 0)) then acc else (gnHexRun s (i + 1) (GenPatch::jpCat acc (GenPatch::jpSpan s i (i + 1))) rest))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: gnIsDump,
  parameters: [s : Bytes, i : Nat],
  result: Bool,
  body: ((GenPatch::jpIsDigit (GenPatch::jpAt s i)) && ((GenPatch::jpIsDigit (GenPatch::jpAt s (i + 6))) && ((GenPatch::jpAt s (i + 7)) =? 32))),
  axioms: [Quot.sound, propext]
}

structure{
  name: GnDump,
  type_parameters: [],
  parameters: [],
  fields: [{name: hex, type: Bytes}, {name: pos, type: Nat}]
}

definition{
  name: gnDumpFrom,
  parameters: [s : Bytes, i : Nat, acc : Bytes, fuel : Nat],
  result: GnDump,
  body: (match fuel
    | Nat.zero => (record GnDump {hex := acc, pos := i})
    | Nat.succ rest => (let at : Nat := (gnSkip s i 80); (if (gnIsDump s at) then (gnDumpFrom s (gnNextLine s at) (gnHexRun s (at + 8) acc 80) rest) else (record GnDump {hex := acc, pos := i})))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: gnPair,
  parameters: [req : Bytes, resp : Bytes],
  result: Bytes,
  body: (GenPatch::jpCat (GenRec::rvPair req resp) 0x0a),
  axioms: [Quot.sound, propext]
}
