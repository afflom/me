\begin{lexlean}{GenSuite}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Foundation.Tar}
\importmodule{GenPatch}
\importmodule{GenRec}
\importmodule{Sha256}
\title{Parsing suite reader}
\begin{semanticmodule}
definition{
  name: jsBody,
  parameters: [archive : Bytes],
  result: Bytes,
  body: (GenPatch::jpSpan archive 512 (512 + (Foundation.Tar::entrySize archive 0))),
  axioms: [Quot.sound, propext]
}

definition{
  name: jsLocal,
  parameters: [archive : Bytes],
  result: (Option Bytes),
  body: (let name : Bytes := (Foundation.Tar::entryName archive 0); (let n : Nat := (GenPatch::jpLen name); (if ((33 <? n) && (GenPatch::jpSame (GenPatch::jpSpan name 0 33) 0x6170702f73706563732f6f7261636c652f6a736f6e2d746573742d73756974652f)) then (ctor Option.some [Bytes] (GenPatch::jpSpan name 33 n)) else (ctor Option.none [Bytes])))),
  axioms: [Quot.sound, propext]
}

definition{
  name: jsSum,
  parameters: [archive : Bytes],
  result: (Option Bytes),
  body: (match (jsLocal archive)
    | Option.none => (ctor Option.none [Bytes])
    | Option.some local => (ctor Option.some [Bytes] (GenPatch::jpCat (Sha256::sha256Hex (jsBody archive)) (GenPatch::jpCat 0x2020 (GenPatch::jpCat local 0x0a))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: jsRefusal,
  parameters: [archive : Bytes],
  result: (Option Bytes),
  body: (match (jsLocal archive)
    | Option.none => (ctor Option.none [Bytes])
    | Option.some local => (ctor Option.some [Bytes] (GenPatch::jpCat (GenRec::rvPair (GenPatch::jpCat 0x63616e6f6e6963616c0a (jsBody archive)) 0x6e6f7420492d4a534f4e) 0x0a))),
  axioms: [Quot.sound, propext]
}

definition{
  name: jsEol,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (if ((Foundation.Tar::byteOr s i) =? 10) then i else (jsEol s (i + 1) rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: jsLines,
  parameters: [s : Bytes, i : Nat, acc : Bytes, fuel : Nat],
  result: Bytes,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (if ((GenPatch::jpLen s) <? (i + 1)) then acc else (let e : Nat := (jsEol s i (prim subtract (GenPatch::jpLen s) i : Nat)); (jsLines s (e + 1) (GenRec::rvJoin acc (GenPatch::jpSpan s i e)) rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: jsVectors,
  parameters: [archive : Bytes, head : Bytes],
  result: Bytes,
  body: (let s : Bytes := (jsBody archive); (GenPatch::jpCat head (GenPatch::jpCat 0x222c226578616d706c6573223a5b0a (GenPatch::jpCat (jsLines s 0 0x ((GenPatch::jpLen s) + 1)) 0x0a5d7d0a)))),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
