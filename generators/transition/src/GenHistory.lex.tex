definition{
  name: ghLine,

  body: (GenPatch::jpCat a (GenPatch::jpCat 0x20 c))

definition{
  name: ghLines,
  parameters: [fc : Bytes, qc : Bytes, tc : Bytes, e : Bytes],
  result: Bytes,
  body: (GenPatch::jpCat (ghLine (GenRec::rvAddr fc) fc) (GenPatch::jpCat 0x0a (GenPatch::jpCat (ghLine (GenRec::rvAddr qc) qc) (GenPatch::jpCat 0x0a (GenPatch::jpCat (ghLine (GenRec::rvAddr tc) tc) (GenPatch::jpCat 0x0a (ghLine (GenRec::rvAddr e) e))))))),

definition{

definition{
  name: ghEmit,
  parameters: [acc : Bytes, on : Bool, req : Bytes, resp : Bytes],
  result: Bytes,
  body: (if (on && ((GenPatch::jpLen req) <? 8193)) then (GenRec::rvJoin acc (GenRec::rvPair req resp)) else acc),
  axioms: [Quot.sound, propext]

definition{
  name: ghHeld,
  parameters: [st : GhState],
  result: Bool,
  body: (0 <? (GenPatch::jpLen (project st entry)))

definition{

definition{
  name: ghMoved,
  parameters: [
    st : GhState,
    emit : Bool,
    mode : Nat,
    d : Bytes,
    q : Bytes,
    fc : Bytes,
    qc : Bytes,
    tc : Bytes
  ],
  result: GhState,
  body: (let chained : Bool := ((ghHeld st) && (GenPatch::jpSame fc (project st after))); (let act : Bytes := (if (chained && true) then (project st act) else 0x); (let pe : Bytes := (if (chained && true) then (project st eaddr) else 0x); (let e : Bytes := (ghEntry (GenRec::snRecord (GenRec::rvAddr fc) (GenRec::rvAddr qc) (GenRec::rvAddr tc) act) pe); (record GhState {acc := (ghOut st emit mode d q fc qc tc chained e pe), act := (GenRec::snTransitionId (GenRec::rvAddr fc) (GenRec::rvAddr qc) act), entry := e, eaddr := (GenRec::rvAddr e), after := tc}))))),

definition{
  name: ghCase,
  parameters: [el : Bytes, st : GhState, emit : Bool, mode : Nat],
  result: GhState,
  body: (match (GenRec::rvMember el 0x646f63)
    | Option.none => st
    | Option.some d => (match (GenRec::rvMember el 0x7061746368)
      | Option.none => st
      | Option.some q => (match (GenJcs::jcCanonicalize d)
        | Option.none => st
        | Option.some fc => (match (GenRec::rvMember el 0x6578706563746564)
          | Option.some x => (match (GenJcs::jcCanonicalize x)
            | Option.none => st
            | Option.some tc => (match (GenJcs::jcCanonicalize q)
              | Option.none => st

  axioms: [Quot.sound, propext]
}

definition{
  name: ghFrom,
  parameters: [cases : Bytes, n : Nat, lo : Nat, hi : Nat, mode : Nat, st : GhState, fuel : Nat],
  result: GhState,
  body: (match fuel
    | Nat.zero => st
    | Nat.succ rest => (match (GenPatch::jpElement cases n)
      | Option.none => st
      | Option.some h => (ghFrom cases (n + 1) lo hi mode (ghCase (GenPatch::jpSpan cases (project h vstart) (project h vend)) st ((lo <? (n + 1)) && (n <? hi)) mode) rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}
