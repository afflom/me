\begin{lexlean}{GenTable}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Foundation.Bytes}
\importmodule{Foundation.Json}
\importmodule{Foundation.Tar}
\importmodule{Foundation.Utf8}
\title{Table generator}
\begin{semanticmodule}
definition{
  name: tableText,
  parameters: [value : String],
  result: Bytes,
  body: (Foundation.Utf8::encode value)
}

definition{
  name: tableSlice,
  parameters: [s : Bytes, start : Nat, count : Nat],
  result: Bytes,
  body: (match (Foundation.Bytes::sliceBytes s start count)
    | Option.none => 0x
    | Option.some b => b)
}

definition{
  name: tableJoin,
  parameters: [parts : (List Bytes)],
  result: Bytes,
  body: (match parts
    | List.nil => 0x
    | List.cons head rest => (Foundation.Bytes::appendBytes head (tableJoin rest))),
  recursive_argument: parts
}

definition{
  name: tableFacet,
  parameters: [target : Bytes],
  result: (Option Bytes),
  body: (let n : Nat := (Foundation.Bytes::byteLength target); (if (n <? 16) then (ctor Option.none [Bytes]) else (if (Foundation.Tar::sameBytes (tableSlice target 0 5) (tableText "site/")) then (if (Foundation.Tar::sameBytes (tableSlice target (prim subtract n 10 : Nat) 10) (tableText "/table.mjs")) then (ctor Option.some [Bytes] (tableSlice target 5 (prim subtract n 15 : Nat))) else (ctor Option.none [Bytes])) else (ctor Option.none [Bytes])))),
  axioms: [propext]
}

definition{
  name: tableSpecPath,
  parameters: [facet : Bytes],
  result: Bytes,
  body: (tableJoin (cons (tableText "app/specs/") (cons facet (cons (tableText "/spec.json") (nil Bytes)))))
}

definition{
  name: tableHas,
  parameters: [spec : Bytes, key : String],
  result: Bool,
  body: (match (Foundation.Json::memberAt spec (Foundation.Json::root spec) (tableText key))
    | Option.none => false
    | Option.some i => true),
  axioms: [Quot.sound, propext]
}

definition{
  name: tableIsFacet,
  parameters: [spec : Bytes],
  result: Bool,
  body: (if (tableHas spec "fields") then (if (tableHas spec "outcomes") then (if (tableHas spec "guarantees") then (tableHas spec "tree") else false) else false) else false),
  axioms: [Quot.sound, propext]
}

definition{
  name: generate,
  parameters: [archive : Bytes, target : Bytes],
  result: (Option Bytes),
  body: (match (tableFacet target)
    | Option.none => (ctor Option.none [Bytes])
    | Option.some facet => (match (Foundation.Tar::find archive (tableSpecPath facet))
      | Option.none => (ctor Option.none [Bytes])
      | Option.some spec => (if (tableIsFacet spec) then (ctor Option.some [Bytes] (tableJoin (cons (tableText "// Written by generators/facets (GenTable) from ") (cons (tableSpecPath facet) (cons (tableText ". Do not edit.\n// The facet's spec as the page and its suites read it. The compiled guest\n// decides; judge walks the same tree for the curator's replay and the\n// suites' cross-check.\n\nexport const SPEC = ") (cons spec (cons (tableText ";\n\n") (cons (tableText "// The fields in the order a request line names them, each with its values.\nexport const FIELDS = SPEC.fields.map((f) => [f.name, f.values]);\nexport const ANSWERS = SPEC.outcomes;\n// The longest request line: every field at its longest value, one space apart.\nexport const MAX_LINE = FIELDS.reduce((n, [, v]) => n + Math.max(...v.map((x) => x.length)), FIELDS.length - 1);\nexport const GUARANTEES = SPEC.guarantees.map((g, i) => ({ id: 'INV-' + String(i + 1).padStart(3, '0'), ...g }));\nexport const SIGNALS = SPEC.signals ?? {};\nexport const TUNABLE = SPEC.tunable ?? [];\nexport const START = SPEC.start ?? {};\nexport const LEARNING = SPEC.learning ?? {};\nexport const PROVIDERS = SPEC.providers;\n\n// The spec's own tree: each test is one field against one value.\nconst walk = (t, r) => (typeof t === 'string' ? t : walk(r[t.if[0]] === t.if[1] ? t.then : t.else, r));\n\nexport function judge(r) {\n  return walk(SPEC.tree, r);\n}\n\nexport function line(r) {\n  return FIELDS.map(([name]) => r[name]).join(' ');\n}\n") (nil Bytes))))))))) else (ctor Option.none [Bytes])))),
  axioms: [Quot.sound, propext]
}

definition{
  name: generateHolon,
  parameters: [archive : Bytes, target : Bytes],
  result: (Option Bytes),
  body: (if (Foundation.Tar::sameBytes target (tableText "site/holon/table.mjs")) then (match (Foundation.Tar::find archive (tableText "app/specs/holon/spec.json"))
    | Option.none => (ctor Option.none [Bytes])
    | Option.some holon => (match (Foundation.Tar::find archive (tableText "app/specs/curate/spec.json"))
      | Option.none => (ctor Option.none [Bytes])
      | Option.some curate => (ctor Option.some [Bytes] (tableJoin (cons (tableText "// Written by generators/facets (GenHolon) from app/specs/holon/spec.json and\n// app/specs/curate/spec.json. Do not edit.\n// The observer's two decisions and the facets it measures, as the page and its\n// suites read them. The compiled guests decide.\n\nconst HOLON_SPEC = ") (cons holon (cons (tableText ";\n\nconst CURATE_SPEC = ") (cons curate (cons (tableText ";\n\n") (cons (tableText "import { SPEC as RESONANCE } from '../resonance/table.mjs';\nimport { PLACES } from '../surface/surface.mjs';\n\n// A facet's tree, as the page and its suites read it: each test is one field\n// against one value.\nconst walk = (t, r) => (typeof t === 'string' ? t : walk(r[t.if[0]] === t.if[1] ? t.then : t.else, r));\nconst decision = (spec) => ({\n  fields: spec.fields.map((f) => [f.name, f.values]),\n  answers: spec.outcomes,\n  maxLine: spec.fields.reduce((n, f) => n + Math.max(...f.values.map((x) => x.length)), spec.fields.length - 1),\n  guarantees: spec.guarantees.map((g, i) => ({ id: 'INV-' + String(i + 1).padStart(3, '0'), ...g })),\n  judge(r) {\n    return walk(spec.tree, r);\n  },\n  line(r) {\n    return this.fields.map(([n]) => r[n]).join(' ');\n  },\n});\n\nexport const HOLON = decision(HOLON_SPEC);\nexport const CURATE = decision(CURATE_SPEC);\n\n// The holon's facets: each with its directive, objective, KPI, floor and the\n// parameters its own spec marks tunable. A place takes its name from the\n// surface model; a facet with a spec takes its title and purpose from it.\nconst SPECS = { resonance: RESONANCE };\nconst named = (s, name) => s.replaceAll('{name}', name);\nexport const FACETS = HOLON_SPEC.facets.flatMap((f) => {\n  if (f.id === 'places') {\n    return PLACES.map((p) => ({ id: p.key, kind: 'place', name: p.name, floor: f.floor, directive: named(f.directive, p.name), objective: named(f.objective, p.name), kpi: named(f.kpi, p.name), tunable: [] }));\n  }\n  if (f.spec && !SPECS[f.spec]) throw new Error('holon: no table for the spec of facet ' + f.id);\n  const spec = f.spec ? SPECS[f.spec] : {};\n  const base = f.id === 'resonance' ? HOLON_SPEC.resonanceDefault : {};\n  return [{ id: f.id, kind: f.spec ? 'facet' : f.id, name: spec.title ?? f.id, floor: f.floor, directive: f.directive || spec.purpose || base.directive, objective: f.objective || base.objective || null, kpi: f.kpi || base.kpi || null, tunable: spec.tunable ?? [] }];\n});\nexport const BANDS = HOLON_SPEC.bands;\nexport const GESTURES = HOLON_SPEC.gestures;\nexport const OBJECTIVES = CURATE_SPEC.objectives;\nexport const RULE30 = { cells: CURATE_SPEC.rule30.cells, seed: CURATE_SPEC.rule30.seed, lanes: CURATE_SPEC.rule30.lanes, vectors: CURATE_SPEC.rule30.vectors };\nexport const CLAIM = CURATE_SPEC.claim;\n") (nil Bytes))))))))))) else (ctor Option.none [Bytes])),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
