\begin{semanticmodule}
definition{
  name: hsEl,
  parameters: [a : Bytes, n : Nat],
  result: Bytes,
  body: (match (PrismSnapshot::jpElement a n)
    | Option.none => 0x
    | Option.some h => (PrismSnapshot::jpSpan a (project h vstart) (project h vend))),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsAddr,
  parameters: [c : Bytes],
  result: Bytes,
  body: (PrismSnapshot::jpCat 0x7368613235363a (Sha256::sha256Hex c)),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsBefore,
  parameters: [t : Bytes],
  result: Bytes,
  body: (PrismSnapshot::jpSpan t 45 116)
}

definition{
  name: hsPatch,
  parameters: [t : Bytes],
  result: Bytes,
  body: (PrismSnapshot::jpSpan t 125 196)
}

definition{
  name: hsAfter,
  parameters: [t : Bytes],
  result: Bytes,
  body: (PrismSnapshot::jpSpan t 205 276)
}

definition{
  name: hsAct,
  parameters: [t : Bytes],
  result: Bytes,
  body: (PrismSnapshot::jpSpan t 287 358)
}

definition{
  name: hsPrevOf,
  parameters: [t : Bytes],
  result: Bytes,
  body: (if (PrismSnapshot::jpSame (PrismSnapshot::snRecord (hsBefore t) (hsPatch t) (hsAfter t) 0x) t) then 0x else (PrismSnapshot::jpSpan t (prim subtract (PrismSnapshot::jpLen t) 85 : Nat) (prim subtract (PrismSnapshot::jpLen t) 14 : Nat))),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsOk,
  parameters: [t : Bytes],
  result: Bool,
  body: (let p : Bytes := (hsPrevOf t); (((PrismSnapshot::snIsAddr (hsBefore t)) && ((PrismSnapshot::snIsAddr (hsPatch t)) && (PrismSnapshot::snIsAddr (hsAfter t)))) && ((((PrismSnapshot::jpLen p) =? 0) || (PrismSnapshot::snIsAddr p)) && (PrismSnapshot::jpSame (PrismSnapshot::snRecord (hsBefore t) (hsPatch t) (hsAfter t) p) t)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsTextOf,
  parameters: [e : Bytes, t : Bytes],
  result: (Option Bytes),
  body: (match (PrismSnapshot::snPrev (hsEl e 1))
    | Option.none => (ctor Option.none [Bytes])
    | Option.some x => (if ((hsOk t) && ((if ((PrismSnapshot::jpLen x) =? 0) then 0 else 1) =? (if ((PrismSnapshot::jpLen (hsPrevOf t)) =? 0) then 0 else 1))) then (ctor Option.some [Bytes] t) else (ctor Option.none [Bytes]))),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsText,
  parameters: [e : Bytes],
  result: (Option Bytes),
  body: (if (((PrismSnapshot::jpCount e) =? 2) && ((PrismSnapshot::jpAt e 1) =? 34)) then (match (PrismSnapshot::jpDecode e 2)
    | Option.none => (ctor Option.none [Bytes])
    | Option.some t => (hsTextOf e t)) else (ctor Option.none [Bytes])),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsPrevEntry,
  parameters: [e : Bytes],
  result: Bytes,
  body: (match (PrismSnapshot::snPrev (hsEl e 1)) | Option.none => 0x | Option.some x => x),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsEntryOf,
  parameters: [r : Bytes, prev : Bytes],
  result: Bytes,
  body: (PrismSnapshot::jpCat 0x5b (PrismSnapshot::jpCat (PrismSnapshot::jpEncode r) (PrismSnapshot::jpCat 0x2c (PrismSnapshot::jpCat (if ((PrismSnapshot::jpLen prev) =? 0) then 0x6e756c6c else (PrismSnapshot::jpCat 0x22 (PrismSnapshot::jpCat prev 0x22))) 0x5d)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsLine,
  parameters: [a : Bytes, c : Bytes],
  result: Bytes,
  body: (PrismSnapshot::jpCat a (PrismSnapshot::jpCat 0x20 c))
}

definition{
  name: hsOut,
  parameters: [r : Bytes, fc : Bytes, qc : Bytes, tc : Bytes, prev : Bytes],
  result: Bytes,
  body: (let e : Bytes := (hsEntryOf r prev); (PrismSnapshot::jpCat (hsLine (hsBefore r) fc) (PrismSnapshot::jpCat 0x0a (PrismSnapshot::jpCat (hsLine (hsPatch r) qc) (PrismSnapshot::jpCat 0x0a (PrismSnapshot::jpCat (hsLine (hsAfter r) tc) (PrismSnapshot::jpCat 0x0a (hsLine (hsAddr e) e)))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsKeepOf,
  parameters: [d : Bytes, q : Bytes, act : Bytes, prev : Bytes],
  result: Bytes,
  body: (let r : Bytes := (PrismSnapshot::snTransitionOf d q act); (if (PrismSnapshot::jpSame (PrismSnapshot::jpSpan r 0 9) 0x646f63756d656e740a) then (match (PrismSnapshot::jcCanonicalize d)
    | Option.none => 0x696e76616c6964
    | Option.some fc => (match (PrismSnapshot::jcCanonicalize q)
      | Option.none => 0x696e76616c6964
      | Option.some qc => (match (PrismSnapshot::jpPatch d q)
        | Option.none => 0x696e76616c6964
        | Option.some x => (match (PrismSnapshot::jcCanonicalize x)
          | Option.none => 0x696e76616c6964
          | Option.some tc => (hsOut r fc qc tc prev))))) else r)),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsKeepOn,
  parameters: [d : Bytes, q : Bytes, v : Bytes],
  result: Bytes,
  body: (match (hsText v)
    | Option.none => 0x696e76616c6964
    | Option.some t => (match (PrismSnapshot::jcCanonicalize d)
      | Option.none => 0x6e6f7420492d4a534f4e
      | Option.some fc => (match (PrismSnapshot::jcCanonicalize v)
        | Option.none => 0x696e76616c6964
        | Option.some vc => (if (PrismSnapshot::jpSame (hsAddr fc) (hsAfter t)) then (hsKeepOf d q (hsAct t) (hsAddr vc)) else 0x74686520646f63756d656e74206973206e6f742074686520686561642773207374617465)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsKeep,
  parameters: [p : Bytes],
  result: Bytes,
  body: (let a : Bytes := (PrismSnapshot::jpCompact p); (if (((PrismSnapshot::jpCount a) =? 3) && (PrismSnapshot::jpValid p)) then (if (PrismSnapshot::jpSame (hsEl a 2) 0x6e756c6c) then (hsKeepOf (hsEl a 0) (hsEl a 1) 0x 0x) else (hsKeepOn (hsEl a 0) (hsEl a 1) (hsEl a 2))) else 0x696e76616c6964)),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsNeeds,
  parameters: [p : Bytes],
  result: Bytes,
  body: (let e : Bytes := (PrismSnapshot::jpCompact p); (if (PrismSnapshot::jpValid p) then (match (hsText e)
    | Option.none => 0x696e76616c6964
    | Option.some t => (PrismSnapshot::jpCat (hsBefore t) (PrismSnapshot::jpCat 0x20 (PrismSnapshot::jpCat (hsPatch t) (PrismSnapshot::jpCat 0x20 (PrismSnapshot::jpCat (hsAfter t) (PrismSnapshot::jpCat 0x20 (if ((PrismSnapshot::jpLen (hsPrevEntry e)) =? 0) then 0x2d else (hsPrevEntry e))))))))) else 0x696e76616c6964)),
  axioms: [Quot.sound, propext]
}

definition{
  name: hsReplay,
  parameters: [p : Bytes],
  result: Bytes,
  body: (let a : Bytes := (PrismSnapshot::jpCompact p); (if (((PrismSnapshot::jpCount a) =? 3) && (PrismSnapshot::jpValid p)) then (match (hsText (hsEl a 2))
    | Option.none => 0x696e76616c6964
    | Option.some t => (PrismSnapshot::snReplayOf (hsEl a 0) (hsEl a 1) (hsPrevOf t) t)) else 0x696e76616c6964)),
  axioms: [Quot.sound, propext]
}

definition{
  name: prismHistoryDispatch,
  parameters: [s : Bytes],
  result: Bytes,
  body: (if (8192 <?   axioms: [Quot.sound, propext]
}
