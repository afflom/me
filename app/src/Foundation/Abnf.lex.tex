definition{
  name: abnfByte,
  parameters: [s : Bytes, i : Nat],
  result: Nat,
  body: (match (prim index s i : (Option UInt8))
    | Option.none => 0
    | Option.some b => (prim to_nat b : Nat)),
  axioms: [Quot.sound, propext]
}

definition{
  name: abnfWidth,
  parameters: [s : Bytes, i : Nat],
  result: Nat,
  body: (let b : Nat := (abnfByte s i); (if (b <? 192) then 1 else (if (b <? 224) then 2 else (if (b <? 240) then 3 else 4)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: abnfTail,
  parameters: [s : Bytes, i : Nat],
  result: Nat,
  body: (prim subtract (abnfByte s i) 128 : Nat),
  axioms: [Quot.sound, propext]
}

definition{
  name: abnfCodeAt,
  parameters: [s : Bytes, i : Nat],
  result: Nat,
  body: (let b : Nat := (abnfByte s i); (if (b <? 192) then b else (if (b <? 224) then ((prim multiply (prim subtract b 192 : Nat) 64 : Nat) + (abnfTail s (i + 1))) else (if (b <? 240) then (((prim multiply (prim subtract b 224 : Nat) 4096 : Nat) + (prim multiply (abnfTail s (i + 1)) 64 : Nat)) + (abnfTail s (i + 2))) else (((prim multiply (prim subtract b 240 : Nat) 262144 : Nat) + (prim multiply (abnfTail s (i + 1)) 4096 : Nat)) + ((prim multiply (abnfTail s (i + 2)) 64 : Nat) + (abnfTail s (i + 3)))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: abnfLower,
  parameters: [c : Nat],
  result: Nat,
  body: (if ((64 <? c) && (c <? 91)) then (c + 32) else c)
}

definition{
  name: abnfHas,
  parameters: [x : Nat, l : (List Nat)],
  result: Bool,
  body: (match l
    | List.nil => false
    | List.cons y rest => (if (x =? y) then true else (abnfHas x rest))),
  recursive_argument: l
}

definition{
  name: abnfAppend,
  parameters: [a : (List Nat), b : (List Nat)],
  result: (List Nat),
  body: (match a | List.nil => b | List.cons x rest => (cons x (abnfAppend rest b))),
  recursive_argument: a
}

definition{
  name: abnfUnion,
  parameters: [a : (List Nat), b : (List Nat)],
  result: (List Nat),
  body: (match b
    | List.nil => a
    | List.cons x rest => (if (abnfHas x a) then (abnfUnion a rest) else (abnfUnion (abnfAppend a (cons x (nil Nat))) rest))),
  recursive_argument: b
}

definition{
  name: abnfFresh,
  parameters: [l : (List Nat), seen : (List Nat)],
  result: (List Nat),
  body: (match l
    | List.nil => (nil Nat)
    | List.cons x rest => (if (abnfHas x seen) then (abnfFresh rest seen) else (cons x (abnfFresh rest seen)))),
  recursive_argument: l
}

definition{
  name: abnfInsertDown,
  parameters: [x : Nat, l : (List Nat)],
  result: (List Nat),
  body: (match l
    | List.nil => (cons x (nil Nat))
    | List.cons y rest => (if (y <? x) then (cons x (cons y rest)) else (cons y (abnfInsertDown x rest)))),
  recursive_argument: l
}

definition{
  name: abnfLongestFirst,
  parameters: [l : (List Nat)],
  result: (List Nat),
  body: (match l
    | List.nil => (nil Nat)
    | List.cons x rest => (abnfInsertDown x (abnfLongestFirst rest))),
  recursive_argument: l
}

definition{
  name: abnfLitEnd,
  parameters: [want : (List Nat), s : Bytes, n : Nat, pos : Nat, sensitive : Bool],
  result: (Option Nat),
  body: (match want
    | List.nil => (ctor Option.some [Nat] pos)
    | List.cons c rest => (if (n <=? pos) then (ctor Option.none [Nat]) else (if (if sensitive then ((abnfCodeAt s pos) =? c) else ((abnfLower (abnfCodeAt s pos)) =? (abnfLower c))) then (abnfLitEnd rest s n (pos + (abnfWidth s pos)) sensitive) else (ctor Option.none [Nat])))),
  axioms: [Quot.sound, propext],
  recursive_argument: want
}

definition{
  name: abnfCount,
  parameters: [l : (List Nat)],
  result: Nat,
  body: (match l | List.nil => 0 | List.cons x rest => ((abnfCount rest) + 1)),
  recursive_argument: l
}

definition{
  name: abnfLitEnds,
  parameters: [want : (List Nat), s : Bytes, n : Nat, sensitive : Bool, at : (List Nat)],
  result: (List Nat),
  body: (match at
    | List.nil => (nil Nat)
    | List.cons p rest => (abnfUnion (match (abnfLitEnd want s n p sensitive)
      | Option.none => (nil Nat)
      | Option.some e => (cons e (nil Nat))) (abnfLitEnds want s n sensitive rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: at
}

definition{
  name: abnfRangeEnds,
  parameters: [low : Nat, high : Nat, s : Bytes, n : Nat, at : (List Nat)],
  result: (List Nat),
  body: (match at
    | List.nil => (nil Nat)
    | List.cons p rest => (abnfUnion (if ((p <? n) && ((low <=? (abnfCodeAt s p)) && ((abnfCodeAt s p) <=? high))) then (cons (p + (abnfWidth s p)) (nil Nat)) else (nil Nat)) (abnfRangeEnds low high s n rest))),
  axioms: [Quot.sound, propext],
  recursive_argument: at
}

definition{

  recursive_argument: a
}

definition{
  name: abnfRule,
  parameters: [rules : (List Node), i : Nat],
  result: Node,
  body: (match rules
    | List.nil => constructor{arguments: [], constructor: Node.Prose, type_arguments: []}
    | List.cons r rest => (if (i =? 0) then r else (abnfRule rest (prim subtract i 1 : Nat)))),
  recursive_argument: rules
}

definition{
  name: abnfEndsGo,
  parameters: [
    fuel : Nat,
    rules : (List Node),
    s : Bytes,
    n : Nat,
    task : EndsTask,
    at : (List Nat)
  ],
  result: (List Nat),
  body: (match fuel
    | Nat.zero => (match task
      | EndsTask.EndsRep low high bounded inner frontier k out seen => (abnfLongestFirst out)
      | EndsTask.EndsOne node => (nil Nat)
      | EndsTask.EndsAlt kids => (nil Nat)
      | EndsTask.EndsSeq kids => (nil Nat))
    | Nat.succ more => (match task
      | EndsTask.EndsOne node => (match node
        | Node.Ref i => (abnfEndsGo more rules s n constructor{
          arguments: [(abnfRule rules i)],
          constructor: EndsTask.EndsOne,
          type_arguments: []
        } at)
        | Node.Lit want => (abnfLitEnds want s n false at)
        | Node.Cs want => (abnfLitEnds want s n true at)
        | Node.Range low high => (abnfRangeEnds low high s n at)
        | Node.Alt kids => (abnfEndsGo more rules s n constructor{
          arguments: [kids],
          constructor: EndsTask.EndsAlt,
          type_arguments: []
        } at)
        | Node.Seq kids => (abnfEndsGo more rules s n constructor{
          arguments: [kids],
          constructor: EndsTask.EndsSeq,
          type_arguments: []
        } at)
        | Node.Rep low high bounded inner => (abnfEndsGo more rules s n constructor{
          arguments: [low, high, bounded, inner, at, 0, (if (low =? 0) then at else (nil Nat)), at],
          constructor: EndsTask.EndsRep,
          type_arguments: []
        } at)
        | Node.Prose => (nil Nat))
      | EndsTask.EndsAlt kids => (match kids
        | List.nil => (nil Nat)
        | List.cons kid rest => (abnfUnion (abnfEndsGo more rules s n constructor{
          arguments: [kid],
          constructor: EndsTask.EndsOne,
          type_arguments: []
        } at) (abnfEndsGo more rules s n constructor{
          arguments: [rest],
          constructor: EndsTask.EndsAlt,
          type_arguments: []
        } at)))
      | EndsTask.EndsSeq kids => (match kids
        | List.nil => at
        | List.cons kid rest => (match at
          | List.nil => (nil Nat)
          | List.cons p others => (abnfEndsGo more rules s n constructor{
            arguments: [rest],
            constructor: EndsTask.EndsSeq,
            type_arguments: []
          } (abnfEndsGo more rules s n constructor{
            arguments: [kid],
            constructor: EndsTask.EndsOne,
            type_arguments: []
          } at))))
      | EndsTask.EndsRep low high bounded inner frontier k out seen => (match frontier
        | List.nil => (abnfLongestFirst out)
        | List.cons p others => (if (bounded && (high <=? k)) then (abnfLongestFirst out) else (let next : (List Nat) := (abnfEndsGo more rules s n constructor{
          arguments: [inner],
          constructor: EndsTask.EndsOne,
          type_arguments: []
        } frontier); (abnfEndsGo more rules s n constructor{
          arguments: [
            low,
            high,
            bounded,
            inner,
            (if ((k + 1) <? low) then next else (abnfFresh next seen)),
            (k + 1),
            (if (low <=? (k + 1)) then (abnfUnion out next) else out),
            (abnfUnion seen next)
          ],
          constructor: EndsTask.EndsRep,
          type_arguments: []
        } at)))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: abnfDeriveGo,
  parameters: [
    fuel : Nat,
    rules : (List Node),
    s : Bytes,
    n : Nat,
    task : DeriveTask,
    pos : Nat,
    stop : Nat,
    depth : Nat
  ],
  result: (Option (List Span)),
  body: (match fuel
    | Nat.zero => (ctor Option.none [(List Span)])
    | Nat.succ more => (match task
      | DeriveTask.DeriveOne node => (if (abnfHas stop (abnfEnds more rules s n node (cons pos (nil Nat)))) then (match node
        | Node.Ref i => (match (abnfDeriveGo more rules s n constructor{
            arguments: [(abnfRule rules i)],
            constructor: DeriveTask.DeriveOne,
            type_arguments: []
          } pos stop (depth + 1))
          | Option.none => (ctor Option.none [(List Span)])
          | Option.some kids => (ctor Option.some [(List Span)] (cons (record Span {rule := i, start := pos, stop := stop, depth := depth}) kids)))
        | Node.Lit want => (ctor Option.some [(List Span)] (nil Span))
        | Node.Cs want => (ctor Option.some [(List Span)] (nil Span))
        | Node.Range low high => (ctor Option.some [(List Span)] (nil Span))
        | Node.Alt kids => (abnfDeriveGo more rules s n constructor{
          arguments: [kids],
          constructor: DeriveTask.DeriveAlt,
          type_arguments: []
        } pos stop depth)
        | Node.Seq kids => (abnfDeriveGo more rules s n constructor{
          arguments: [kids],
          constructor: DeriveTask.DeriveSeq,
          type_arguments: []
        } pos stop depth)
        | Node.Rep low high bounded inner => (abnfDeriveGo more rules s n constructor{
          arguments: [low, high, bounded, inner, 0],
          constructor: DeriveTask.DeriveRep,
          type_arguments: []
        } pos stop depth)
        | Node.Prose => (ctor Option.none [(List Span)])) else (ctor Option.none [(List Span)]))
      | DeriveTask.DeriveAlt kids => (match kids
        | List.nil => (ctor Option.none [(List Span)])
        | List.cons kid rest => (match (abnfDeriveGo more rules s n constructor{
            arguments: [kid],
            constructor: DeriveTask.DeriveOne,
            type_arguments: []
          } pos stop depth)
          | Option.none => (abnfDeriveGo more rules s n constructor{
            arguments: [rest],
            constructor: DeriveTask.DeriveAlt,
            type_arguments: []
          } pos stop depth)
          | Option.some found => (ctor Option.some [(List Span)] found)))
      | DeriveTask.DeriveSeq kids => (match kids
        | List.nil => (if (pos =? stop) then (ctor Option.some [(List Span)] (nil Span)) else (ctor Option.none [(List Span)]))
        | List.cons kid rest => (abnfDeriveGo more rules s n constructor{
          arguments: [kid, rest, (abnfEnds more rules s n kid (cons pos (nil Nat)))],
          constructor: DeriveTask.DeriveSeqTry,
          type_arguments: []
        } pos stop depth))
      | DeriveTask.DeriveSeqTry kid rest cands => (match cands
        | List.nil => (ctor Option.none [(List Span)])
        | List.cons e others => (if (stop <? e) then (abnfDeriveGo more rules s n constructor{
          arguments: [kid, rest, others],
          constructor: DeriveTask.DeriveSeqTry,
          type_arguments: []
        } pos stop depth) else (match (abnfDeriveGo more rules s n constructor{
            arguments: [kid],
            constructor: DeriveTask.DeriveOne,
            type_arguments: []
          } pos e depth)
          | Option.none => (abnfDeriveGo more rules s n constructor{
            arguments: [kid, rest, others],
            constructor: DeriveTask.DeriveSeqTry,
            type_arguments: []
          } pos stop depth)
          | Option.some head => (match (abnfDeriveGo more rules s n constructor{
              arguments: [rest],
              constructor: DeriveTask.DeriveSeq,
              type_arguments: []
            } e stop depth)
            | Option.none => (abnfDeriveGo more rules s n constructor{
              arguments: [kid, rest, others],
              constructor: DeriveTask.DeriveSeqTry,
              type_arguments: []
            } pos stop depth)
            | Option.some tail => (ctor Option.some [(List Span)] (abnfJoin head tail))))))
      | DeriveTask.DeriveRep low high bounded inner k => (if ((pos =? stop) && (low <=? k)) then (ctor Option.some [(List Span)] (nil Span)) else (if (bounded && (high <=? k)) then (ctor Option.none [(List Span)]) else (abnfDeriveGo more rules s n constructor{
        arguments: [
          low,
          high,
          bounded,
          inner,
          (abnfEnds more rules s n inner (cons pos (nil Nat))),
          k
        ],
        constructor: DeriveTask.DeriveRepTry,
        type_arguments: []
      } pos stop depth)))
      | DeriveTask.DeriveRepTry low high bounded inner cands k => (match cands
        | List.nil => (ctor Option.none [(List Span)])
        | List.cons e others => (if ((stop <? e) || ((e =? pos) && (low <=? k))) then (abnfDeriveGo more rules s n constructor{
          arguments: [low, high, bounded, inner, others, k],
          constructor: DeriveTask.DeriveRepTry,
          type_arguments: []
        } pos stop depth) else (match (abnfDeriveGo more rules s n constructor{
            arguments: [inner],
            constructor: DeriveTask.DeriveOne,
            type_arguments: []
          } pos e depth)
          | Option.none => (abnfDeriveGo more rules s n constructor{
            arguments: [low, high, bounded, inner, others, k],
            constructor: DeriveTask.DeriveRepTry,
            type_arguments: []
          } pos stop depth)
          | Option.some head => (match (abnfDeriveGo more rules s n constructor{
              arguments: [low, high, bounded, inner, (k + 1)],
              constructor: DeriveTask.DeriveRep,
              type_arguments: []
            } e stop depth)
            | Option.none => (abnfDeriveGo more rules s n constructor{
              arguments: [low, high, bounded, inner, others, k],
              constructor: DeriveTask.DeriveRepTry,
              type_arguments: []
            } pos stop depth)
            | Option.some tail => (ctor Option.some [(List Span)] (abnfJoin head tail)))))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: abnfDecodeGo,
  parameters: [fuel : Nat, many : Bool, count : Nat, b : Bytes, at : Nat],
  result: DecodedList,
  body: (match fuel
    | Nat.zero => (record DecodedList {nodes := (if many then (nil Node) else (cons constructor{
      arguments: [],
      constructor: Node.Prose,
      type_arguments: []
    } (nil Node))), next := at})
    | Nat.succ more => (if many then (if (count =? 0) then (record DecodedList {nodes := (nil Node), next := at}) else (let head : DecodedList := (abnfDecodeGo more false 0 b at); (let rest : DecodedList := (abnfDecodeGo more true (prim subtract count 1 : Nat) b (project head next)); (record DecodedList {nodes := (cons (abnfFirst (project head nodes)) (project rest nodes)), next := (project rest next)})))) else (let tag : Nat := (abnfByte b at); (if (tag =? 0) then (record DecodedList {nodes := (cons constructor{
      arguments: [(abnfWord b (at + 1))],
      constructor: Node.Ref,
      type_arguments: []
    } (nil Node)), next := (at + 3)}) else (if ((tag =? 1) || (tag =? 2)) then (let got : DecodedCodes := (abnfCodes (abnfWord b (at + 1)) b (at + 3)); (record DecodedList {nodes := (cons (if (tag =? 1) then constructor{
      arguments: [(project got codes)],
      constructor: Node.Lit,
      type_arguments: []
    } else constructor{
      arguments: [(project got codes)],
      constructor: Node.Cs,
      type_arguments: []
    }) (nil Node)), next := (project got next)})) else (if (tag =? 3) then (record DecodedList {nodes := (cons constructor{
      arguments: [(abnfTriple b (at + 1)), (abnfTriple b (at + 4))],
      constructor: Node.Range,
      type_arguments: []
    } (nil Node)), next := (at + 7)}) else (if ((tag =? 4) || (tag =? 5)) then (let got : DecodedList := (abnfDecodeGo more true (abnfWord b (at + 1)) b (at + 3)); (record DecodedList {nodes := (cons (if (tag =? 4) then constructor{
      arguments: [(project got nodes)],
      constructor: Node.Alt,
      type_arguments: []
    } else constructor{
      arguments: [(project got nodes)],
      constructor: Node.Seq,
      type_arguments: []
    }) (nil Node)), next := (project got next)})) else (if (tag =? 6) then (let inner : DecodedList := (abnfDecodeGo more false 0 b (at + 6)); (record DecodedList {nodes := (cons constructor{
      arguments: [
        (abnfWord b (at + 1)),
        (abnfWord b (at + 3)),
        ((abnfByte b (at + 5)) =? 1),
        (abnfFirst (project inner nodes))
      ],
      constructor: Node.Rep,
      type_arguments: []
    } (nil Node)), next := (project inner next)})) else (record DecodedList {nodes := (cons constructor{
      arguments: [],
      constructor: Node.Prose,
      type_arguments: []
    } (nil Node)), next := (at + 1)}))))))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

  parameters: [x : Span],
  result: String,
  body: (prim join (cons (abnfDecimal (project x rule)) (cons (abnfDecimal (project x start)) (cons (abnfDecimal (project x stop)) (cons (abnfDecimal (project x depth)) (nil String))))) "," : String),
  axioms: [propext]
}

definition{
  name: abnfSpansText,
  parameters: [spans : (List Span)],
  result: (List String),
  body: (match spans
    | List.nil => (nil String)
    | List.cons x rest => (cons (abnfSpanText x) (abnfSpansText rest))),
  axioms: [propext],
  recursive_argument: spans
}

definition{
  name: abnfNumberEnd,
  parameters: [fuel : Nat, s : Bytes, i : Nat, acc : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ more => (let d : Nat := (abnfByte s i); (if ((d <? 48) || (57 <? d)) then acc else (abnfNumberEnd more s (i + 1) ((prim multiply acc 10 : Nat) + (prim subtract d 48 : Nat)))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: abnfDigitsCount,
  parameters: [fuel : Nat, s : Bytes, i : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ more => (let d : Nat := (abnfByte s i); (if (((d <? 48) || (57 <? d)) || ((prim length s : Nat) <=? i)) then i else (abnfDigitsCount more s (i + 1))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: abnfAnswer,
  parameters: [table : Bytes, request : Bytes, fuel : Nat],
  result: Bytes,
  body: (let cut : Nat := (abnfDigitsCount 6 request 0); (if ((cut =? 0) || (not ((abnfByte request cut) =? 32))) then (prim utf8_encode "invalid" : Bytes) else (match (prim slice request (cut + 1) (prim subtract (prim length request : Nat) (cut + 1) : Nat) : (Option Bytes))
    | Option.none => (prim utf8_encode "invalid" : Bytes)
    | Option.some text => (match (abnfMatch (abnfTable table) (abnfNumberEnd 6 request 0 0) text (prim length text : Nat) fuel)
      | Option.none => (prim utf8_encode "none" : Bytes)
      | Option.some spans => (prim utf8_encode (prim join (abnfSpansText spans) " " : String) : Bytes))))),
  axioms: [Quot.sound, propext]
}
