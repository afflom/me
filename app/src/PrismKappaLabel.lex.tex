definition{
  name: kappaLabelByteAt,
  parameters: [s : Bytes, i : Nat],
  result: (Option Nat),
  body: (match (prim index s i : (Option UInt8))
    | Option.none => (ctor Option.none [Nat])
    | Option.some b => (ctor Option.some [Nat] (prim to_nat b : Nat))),
  axioms: [Quot.sound, propext]
}

definition{
  name: kappaLabelSliceOf,
  parameters: [s : Bytes, start : Nat, count : Nat],
  result: Bytes,
  body: (match (prim slice s start count : (Option Bytes))
    | Option.none => 0x
    | Option.some part => part)
}

definition{
  name: kappaLabelStopAt,
  parameters: [s : Bytes, stop : Nat, i : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (match (kappaLabelByteAt s i)
      | Option.none => i
      | Option.some d => (if (d =? stop) then i else (kappaLabelStopAt s stop (i + 1) rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: kappaLabelAllHex,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Bool,
  body: (match fuel
    | Nat.zero => true
    | Nat.succ rest => (match (kappaLabelByteAt s i)
      | Option.none => true
      | Option.some d => (if ((d <? 48) || (102 <? d)) then false else (if ((57 <? d) && (d <? 97)) then false else (kappaLabelAllHex s (i + 1) rest))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: kappaLabelAxisOf,
  parameters: [value : Bytes],
  result: Nat,

    | Option.none => (PrismKappa::prismKappaAxisOther)
    | Option.some v => v)

definition{
  name: kappaLabelReadLabel,
  parameters: [label : Bytes, digest : Nat],
  result: PrismKappa::PrismKappaRequest,
  body: (let n : Nat := (Foundation.Bytes::byteLength label); (let c : Nat := (kappaLabelStopAt label 58 0 n); (if (c =? n) then (record PrismKappa::PrismKappaRequest {colon := (PrismKappa::prismKappaColonAbsent), axis := (PrismKappa::prismKappaAxisOther), digits := (PrismKappa::prismKappaDigitsWrong), hex := (PrismKappa::prismKappaHexBad), digest := digest}) else (let a : Nat := (kappaLabelAxisOf (kappaLabelSliceOf label 0 c)); (record PrismKappa::PrismKappaRequest {colon := (PrismKappa::prismKappaColonPresent), axis := a, digits := (if ((prim subtract n (c + 1) : Nat) =? (kappaLabelHexLength a)) then (PrismKappa::prismKappaDigitsRight) else (PrismKappa::prismKappaDigitsWrong)), hex := (if (kappaLabelAllHex label (c + 1) n) then (PrismKappa::prismKappaHexLower) else (PrismKappa::prismKappaHexBad)), digest := digest}))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: kappaLabelDecode,
  parameters: [s : Bytes],

  body: (let n : Nat := (Foundation.Bytes::byteLength s); (let e : Nat := (kappaLabelStopAt s 32 0 n); (if (e <? n) then (match (kappaLabelParseDigest (kappaLabelSliceOf s (e + 1) (prim subtract n (e + 1) : Nat)))

  axioms: [Quot.sound, propext]
}
