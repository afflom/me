definition{
  name: holonRegisterStart,
  parameters: [s : Bytes, i : Nat, k : Nat],
  result: Nat,
  body: (match k
    | Nat.zero => i
    | Nat.succ rest => (holonRegisterStart s ((PrismHolon::prismHolonFieldEnd s i (Foundation.Bytes::byteLength s)) + 1) rest)),
  axioms: [Quot.sound, propext],
  recursive_argument: k
}

definition{
  name: holonRegisterWord,
  parameters: [s : Bytes, k : Nat],
  result: Bytes,
  body: (let a : Nat := (holonRegisterStart s 0 k); (PrismHolon::prismHolonSliceOf s a (prim subtract (PrismHolon::prismHolonFieldEnd s a (Foundation.Bytes::byteLength s)) a : Nat))),
  axioms: [Quot.sound, propext]
}

definition{
  name: holonRegisterNat,
  parameters: [s : Bytes, i : Nat, acc : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ rest => (match (PrismHolon::prismHolonByteAt s i)
      | Option.none => acc
      | Option.some d => (if (d =? 32) then acc else (holonRegisterNat s (i + 1) ((prim multiply acc 16 : Nat) + (if (d <? 58) then (prim subtract d 48 : Nat) else (prim subtract d 87 : Nat))) rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: holonRegisterDigitsFrom,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Bool,
  body: (match fuel
    | Nat.zero => (match (PrismHolon::prismHolonByteAt s i)
      | Option.none => true
      | Option.some d => (d =? 32))
    | Nat.succ rest => (match (PrismHolon::prismHolonByteAt s i)
      | Option.none => true
      | Option.some d => (if (d =? 32) then true else ((((47 <? d) && (d <? 58)) || ((96 <? d) && (d <? 103))) && (holonRegisterDigitsFrom s (i + 1) rest))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: holonRegisterDigits,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Bool,
  body: (match (PrismHolon::prismHolonByteAt s i)
    | Option.none => false
    | Option.some d => ((if (d =? 32) then false else true) && (holonRegisterDigitsFrom s i fuel))),
  axioms: [Quot.sound, propext]
}

definition{
  name: holonRegisterPow,

  recursive_argument: k

definition{
  name: holonRegisterNext,
  parameters: [s : Nat, i : Nat, fuel : Nat],

    | Nat.zero => 0
    | Nat.succ rest => ((holonRegisterCell s i) + (holonRegisterNext s (i + 1) rest))),
