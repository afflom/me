((project gate surface) + 1))) || ((((browserGateWaits (project gate surface) 0 0) && (browserGateWaits (project gate surface) 0 1)) && (browserGateWaits (project gate surface) 1 0)) && (browserGateWaits (project gate surface) 1 1))))
